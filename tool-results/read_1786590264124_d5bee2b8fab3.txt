   339→// ============================ SIDEBAR ============================
   340→function SidebarContent({
   341→  user, cardsCount, maxCards, unreadCount, activeSection, onNavigate, onLogout, onSupport,
   342→  onShareCard,
   343→}: {
   344→  user: { name: string; email: string; plan: PlanType; id?: string };
   345→  cardsCount: number;
   346→  maxCards: number;
   347→  unreadCount: number;
   348→  activeSection: SectionId;
   349→  onNavigate: (id: SectionId) => void;
   350→  onLogout: () => void;
   351→  onSupport: () => void;
   352→  onShareCard: (card: BusinessCard | null) => void;
   353→}) {
   354→  const planColors = PLAN_COLORS[user.plan];
   355→  const plan = PLANS[user.plan];
   356→  const cardsUsagePercent = Math.min(100, (cardsCount / maxCards) * 100);
   357→  // Storage is approximated for the ring (we use plan.storage and assume 35% used as a demo)
   358→  const storageUsedPct = Math.min(100, 35);
   359→  const primaryCard = useAppStore(s => {
   360→    const all = s.cards;
   361→    return all.find(c => c.userId === user.id) || null;
   362→  });
   363→  const favoriteIds = useAppStore(s => s.favoriteCardIds);
   364→  const unreadNotifications = useAppStore(s =>
   365→    s.notifications.filter(n => !n.read).length
   366→  );
   367→
   368→  // SVG circular progress ring geometry
   369→  const ringSize = 64;
   370→  const ringStroke = 6;
   371→  const radius = (ringSize - ringStroke) / 2;
   372→  const circumference = 2 * Math.PI * radius;
   373→  const cardsDash = (cardsUsagePercent / 100) * circumference;
   374→
   375→  const handlePublicCard = () => {
   376→    if (primaryCard) {
   377→      useAppStore.getState().selectCard(primaryCard.id);
   378→      useAppStore.getState().navigate('public-card');
   379→    } else {
   380→      toast.info('No tienes tarjetas para mostrar');
   381→    }
   382→  };
   383→  const handleShare = () => {
   384→    onShareCard(primaryCard);
   385→  };
   386→  const handleHelp = () => {
   387→    toast.info('Centro de ayuda', { description: 'Contáctanos en soporte@ftpdigitalplus.com' });
   388→  };
   389→
   390→  return (
   391→    <div className="flex h-full flex-col">
   392→      {/* Logo + theme toggle */}
   393→      <div className="flex items-center justify-between border-b px-5 py-4">
   394→        <FTPLogo className="h-9 w-auto" />
   395→        <ThemeToggle className="text-slate-600 hover:bg-emerald-50 hover:text-emerald-700" />
   396→      </div>
   397→
   398→      {/* User info + plan progress ring */}
   399→      <div className="border-b px-5 py-4">
   400→        <div className="flex items-center gap-3">
   401→          <div className="relative">
   402→            <Avatar className="h-11 w-11 ring-2 ring-emerald-200">
   403→              <AvatarFallback className="bg-gradient-to-br from-emerald-500 to-emerald-700 text-sm font-bold text-white">
   404→                {user.name.split(' ').map(n => n[0]).slice(0, 2).join('').toUpperCase()}
   405→              </AvatarFallback>
   406→            </Avatar>
   407→            {/* Online status indicator */}
   408→            <span
   409→              className="absolute -bottom-0.5 -right-0.5 h-3.5 w-3.5 rounded-full border-2 border-white bg-emerald-500"
   410→              aria-label="En línea"
   411→            >
   412→              <span className="absolute inset-0 animate-ping rounded-full bg-emerald-400 opacity-75" />
   413→            </span>
   414→          </div>
   415→          <div className="min-w-0 flex-1">
   416→            <p className="truncate text-sm font-semibold text-slate-800">{user.name}</p>
   417→            <p className="truncate text-xs text-muted-foreground">{user.email}</p>
   418→          </div>
   419→        </div>
   420→
   421→        {/* Plan progress ring + usage */}
   422→        <div className="mt-4 flex items-center gap-3 rounded-xl bg-gradient-to-br from-slate-50 to-emerald-50/40 p-3 ring-1 ring-slate-200/60">
   423→          {/* Circular progress ring (SVG) */}
   424→          <div className="relative" style={{ width: ringSize, height: ringSize }}>
   425→            <svg width={ringSize} height={ringSize} className="-rotate-90">
   426→              <circle
   427→                cx={ringSize / 2}
   428→                cy={ringSize / 2}
   429→                r={radius}
   430→                fill="none"
   431→                stroke="currentColor"
   432→                strokeWidth={ringStroke}
   433→                className="text-slate-200"
   434→              />
   435→              <circle
   436→                cx={ringSize / 2}
   437→                cy={ringSize / 2}
   438→                r={radius}
   439→                fill="none"
   440→                stroke="url(#planGrad)"
   441→                strokeWidth={ringStroke}
   442→                strokeDasharray={`${cardsDash} ${circumference}`}
   443→                strokeLinecap="round"
   444→                className="transition-all duration-500"
   445→              />
   446→              <defs>
   447→                <linearGradient id="planGrad" x1="0" y1="0" x2="1" y2="1">
   448→                  <stop offset="0%" stopColor="#059669" />
   449→                  <stop offset="100%" stopColor="#f59e0b" />
   450→                </linearGradient>
   451→              </defs>
   452→            </svg>
   453→            <div className="absolute inset-0 flex items-center justify-center">
   454→              <span className="text-xs font-bold text-slate-700">{cardsUsagePercent.toFixed(0)}%</span>
   455→            </div>
   456→          </div>
   457→          <div className="min-w-0 flex-1">
   458→            <p className="text-xs font-semibold text-slate-700">Uso del plan</p>
   459→            <p className="text-[11px] text-muted-foreground">
   460→              <CreditCard className="mr-1 inline h-3 w-3" />
   461→              {cardsCount} / {maxCards} tarjetas
   462→            </p>
   463→            <p className="text-[11px] text-muted-foreground">
   464→              <Database className="mr-1 inline h-3 w-3" />

   472→          {user.plan === 'pro' && <Crown className="h-3 w-3" />}
   473→          {user.plan === 'basico' && <Zap className="h-3 w-3" />}
   474→          {user.plan === 'gratis' && <Sparkles className="h-3 w-3" />}
   475→          Plan {planColors.label}
   476→        </div>
   477→      </div>
   478→
   479→      {/* Nav */}
   480→      <nav className="flex-1 overflow-y-auto px-3 py-4 custom-scrollbar">
   481→        <p className="px-3 pb-2 text-[10px] font-bold uppercase tracking-wider text-muted-foreground/70">
   482→          Menú Principal
   483→        </p>
   484→        <ul className="space-y-1">
   485→          {DASHBOARD_SECTIONS.map((section) => {
   486→            const isActive = activeSection === section.id;
   487→            const isMessages = section.id === 'messages';
   488→            const isNotifications = section.id === 'notifications';
   489→            return (
   490→              <li key={section.id}>
   491→                <button
   492→                  onClick={() => onNavigate(section.id as SectionId)}
   493→                  className={cn(
   494→                    'group flex w-full items-center gap-3 rounded-lg px-3 py-2.5 text-sm font-medium transition-all',
   495→                    isActive
   496→                      ? 'bg-emerald-600 text-white shadow-sm shadow-emerald-600/20'
   497→                      : 'text-slate-600 hover:bg-emerald-50 hover:text-emerald-700'

   695→function getRelativeTime(iso: string): string {
   696→  const now = Date.now();
   697→  const diff = now - new Date(iso).getTime();
   698→  const minutes = Math.floor(diff / (1000 * 60));
   699→  if (minutes < 1) return 'hace un momento';
   700→  if (minutes < 60) return `hace ${minutes} min`;
   701→  const hours = Math.floor(minutes / 60);
   702→  if (hours < 24) return `hace ${hours} h`;
   703→  const days = Math.floor(hours / 24);
   704→  if (days === 1) return 'ayer';
   705→  return `hace ${days} días`;
   706→}
