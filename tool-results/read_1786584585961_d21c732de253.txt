   105→  const plan = PLANS[currentUser.plan];
   106→  const unreadCount = messages.filter(m => !m.read).length;

   111→      navigate('stats');

   125→    }

   127→      navigate('template-gallery');

   129→      return;

   180→          <Button
   181→            variant="ghost"
   182→            size="icon"

   184→            onClick={() => window.dispatchEvent(new CustomEvent('ftp:open-global-search'))}
   185→            className="text-slate-600 hover:bg-emerald-50 hover:text-emerald-700"
   186→          >
   187→            <Search className="h-5 w-5" />

   221→                initial={{ opacity: 0, y: 10 }}

   224→                transition={{ duration: 0.25 }}

   226→                {activeSection === 'tablero' && (
   227→                  <TableroSection
   228→                    onCreateOpen={() => setCreateOpen(true)}
   229→                    onShareCard={openShare}
   230→                  />
   231→                )}
   232→                {activeSection === 'messages' && <MessagesSection />}
   233→                {activeSection === 'appointments' && <AppointmentsSection />}

   271→// ============================ SIDEBAR ============================
   272→function SidebarContent({
   273→  user, cardsCount, maxCards, unreadCount, activeSection, onNavigate, onLogout, onSupport,
   274→  onShareCard,
   275→}: {
   276→  user: { name: string; email: string; plan: PlanType; id?: string };
   277→  cardsCount: number;
   278→  maxCards: number;
   279→  unreadCount: number;
   280→  activeSection: SectionId;
   281→  onNavigate: (id: SectionId) => void;
   282→  onLogout: () => void;
   283→  onSupport: () => void;
   284→  onShareCard: (card: BusinessCard | null) => void;
   285→}) {
   286→  const planColors = PLAN_COLORS[user.plan];
   287→  const plan = PLANS[user.plan];
   288→  const cardsUsagePercent = Math.min(100, (cardsCount / maxCards) * 100);
   289→  // Storage is approximated for the ring (we use plan.storage and assume 35% used as a demo)
   290→  const storageUsedPct = Math.min(100, 35);
   291→  const primaryCard = useAppStore(s => {
   292→    const all = s.cards;
   293→    return all.find(c => c.userId === user.id) || null;
   294→  });
   295→  const favoriteIds = useAppStore(s => s.favoriteCardIds);
   296→  const unreadNotifications = useAppStore(s =>
   297→    s.notifications.filter(n => !n.read).length
   298→  );
   299→
   300→  // SVG circular progress ring geometry
   301→  const ringSize = 64;
   302→  const ringStroke = 6;
   303→  const radius = (ringSize - ringStroke) / 2;
   304→  const circumference = 2 * Math.PI * radius;
   305→  const cardsDash = (cardsUsagePercent / 100) * circumference;
   306→
   307→  const handlePublicCard = () => {
   308→    if (primaryCard) {
   309→      useAppStore.getState().selectCard(primaryCard.id);
   310→      useAppStore.getState().navigate('public-card');
   311→    } else {
   312→      toast.info('No tienes tarjetas para mostrar');
   313→    }
   314→  };
   315→  const handleShare = () => {
   316→    onShareCard(primaryCard);
   317→  };
   318→  const handleHelp = () => {
   319→    toast.info('Centro de ayuda', { description: 'Contáctanos en soporte@ftpdigitalplus.com' });
   320→  };
   321→
   322→  return (
   323→    <div className="flex h-full flex-col">
   324→      {/* Logo + theme toggle */}
   325→      <div className="flex items-center justify-between border-b px-5 py-4">
   326→        <FTPLogo className="h-9 w-auto" />
   327→        <ThemeToggle className="text-slate-600 hover:bg-emerald-50 hover:text-emerald-700" />
   328→      </div>
   329→
   330→      {/* User info + plan progress ring */}
   331→      <div className="border-b px-5 py-4">
   332→        <div className="flex items-center gap-3">
   333→          <div className="relative">
   334→            <Avatar className="h-11 w-11 ring-2 ring-emerald-200">
   335→              <AvatarFallback className="bg-gradient-to-br from-emerald-500 to-emerald-700 text-sm font-bold text-white">
   336→                {user.name.split(' ').map(n => n[0]).slice(0, 2).join('').toUpperCase()}
   337→              </AvatarFallback>
   338→            </Avatar>
   339→            {/* Online status indicator */}
   340→            <span
   341→              className="absolute -bottom-0.5 -right-0.5 h-3.5 w-3.5 rounded-full border-2 border-white bg-emerald-500"
   342→              aria-label="En línea"
   343→            >
   344→              <span className="absolute inset-0 animate-ping rounded-full bg-emerald-400 opacity-75" />
   345→            </span>
   346→          </div>
   347→          <div className="min-w-0 flex-1">
   348→            <p className="truncate text-sm font-semibold text-slate-800">{user.name}</p>
   349→            <p className="truncate text-xs text-muted-foreground">{user.email}</p>
   350→          </div>
   351→        </div>
   352→
   353→        {/* Plan progress ring + usage */}
   354→        <div className="mt-4 flex items-center gap-3 rounded-xl bg-gradient-to-br from-slate-50 to-emerald-50/40 p-3 ring-1 ring-slate-200/60">
   355→          {/* Circular progress ring (SVG) */}
   356→          <div className="relative" style={{ width: ringSize, height: ringSize }}>
   357→            <svg width={ringSize} height={ringSize} className="-rotate-90">
   358→              <circle
   359→                cx={ringSize / 2}
   360→                cy={ringSize / 2}
   361→                r={radius}
   362→                fill="none"
   363→                stroke="currentColor"
   364→                strokeWidth={ringStroke}
   365→                className="text-slate-200"
   366→              />
   367→              <circle
   368→                cx={ringSize / 2}
   369→                cy={ringSize / 2}
   370→                r={radius}
   371→                fill="none"
   372→                stroke="url(#planGrad)"
   373→                strokeWidth={ringStroke}
   374→                strokeDasharray={`${cardsDash} ${circumference}`}
   375→                strokeLinecap="round"
   376→                className="transition-all duration-500"
   377→              />
   378→              <defs>
   379→                <linearGradient id="planGrad" x1="0" y1="0" x2="1" y2="1">
   380→                  <stop offset="0%" stopColor="#059669" />
   381→                  <stop offset="100%" stopColor="#f59e0b" />
   382→                </linearGradient>
   383→              </defs>
   384→            </svg>
   385→            <div className="absolute inset-0 flex items-center justify-center">
   386→              <span className="text-xs font-bold text-slate-700">{cardsUsagePercent.toFixed(0)}%</span>
   387→            </div>
   388→          </div>
   389→          <div className="min-w-0 flex-1">
   390→            <p className="text-xs font-semibold text-slate-700">Uso del plan</p>
   391→            <p className="text-[11px] text-muted-foreground">
   392→              <CreditCard className="mr-1 inline h-3 w-3" />
   393→              {cardsCount} / {maxCards} tarjetas
   394→            </p>
   395→            <p className="text-[11px] text-muted-foreground">
   396→              <Database className="mr-1 inline h-3 w-3" />
   397→              {storageUsedPct}% de {plan.storage} MB
   398→            </p>
   399→          </div>
   400→        </div>
   401→
   402→        {/* Plan badge */}
   403→        <div className={cn('mt-3 inline-flex items-center gap-1 rounded-full px-2.5 py-1 text-xs font-semibold ring-1', planColors.bg, planColors.text, planColors.ring)}>
   404→          {user.plan === 'pro' && <Crown className="h-3 w-3" />}
   405→          {user.plan === 'basico' && <Zap className="h-3 w-3" />}
   406→          {user.plan === 'gratis' && <Sparkles className="h-3 w-3" />}
   407→          Plan {planColors.label}
   408→        </div>
   409→      </div>
   410→
   411→      {/* Nav */}
   412→      <nav className="flex-1 overflow-y-auto px-3 py-4 custom-scrollbar">
   413→        <p className="px-3 pb-2 text-[10px] font-bold uppercase tracking-wider text-muted-foreground/70">
   414→          Menú Principal
   415→        </p>
   416→        <ul className="space-y-1">
   417→          {DASHBOARD_SECTIONS.map((section) => {
   418→            const isActive = activeSection === section.id;
   419→            const isMessages = section.id === 'messages';
   420→            const isNotifications = section.id === 'notifications';

   422→              <li key={section.id}>
   423→                <button
   424→                  onClick={() => onNavigate(section.id as SectionId)}
   425→                  className={cn(
   426→                    'group flex w-full items-center gap-3 rounded-lg px-3 py-2.5 text-sm font-medium transition-all',
   427→                    isActive
   428→                      ? 'bg-emerald-600 text-white shadow-sm shadow-emerald-600/20'
   429→                      : 'text-slate-600 hover:bg-emerald-50 hover:text-emerald-700'
   430→                  )}
   431→                >
   432→                  <DynamicIcon name={section.icon} className={cn('h-[18px] w-[18px] shrink-0', isActive ? 'text-white' : 'text-slate-500 group-hover:text-emerald-600')} />
   433→                  <span className="flex-1 text-left">{section.name}</span>
   434→                  {isMessages && unreadCount > 0 && (
   435→                    <span className={cn(
   436→                      'rounded-full px-1.5 py-0.5 text-[10px] font-bold',
   437→                      isActive ? 'bg-white text-emerald-700' : 'bg-amber-500 text-white'
   438→                    )}>
   439→                      {unreadCount}
   440→                    </span>
   441→                  )}
   442→                  {isNotifications && unreadNotifications > 0 && (
   443→                    <span className={cn(
   444→                      'rounded-full px-1.5 py-0.5 text-[10px] font-bold',
   445→                      isActive ? 'bg-white text-emerald-700' : 'bg-rose-500 text-white'
   446→                    )}>
   447→                      {unreadNotifications > 9 ? '9+' : unreadNotifications}
   448→                    </span>
   449→                  )}
   450→                  {section.id === 'tablero' && favoriteIds.length > 0 && !isActive && (
   451→                    <Star className="h-3.5 w-3.5 fill-amber-400 text-amber-400" />
   452→                  )}
   453→                  {isActive && <ChevronRight className="h-4 w-4" />}
   454→                </button>
   455→              </li>
   456→            );
   457→          })}
   458→        </ul>
   459→      </nav>
   460→
   461→      {/* Quick action buttons */}
   462→      <div className="border-t px-3 py-3">
   463→        <p className="px-2 pb-2 text-[10px] font-bold uppercase tracking-wider text-muted-foreground/70">
   464→          Acciones Rápidas
   465→        </p>
   466→        <div className="grid grid-cols-3 gap-1.5">
   467→          <TooltipProvider>
   468→            <Tooltip>
   469→              <TooltipTrigger asChild>
   470→                <Button
   471→                  variant="outline"
   472→                  size="sm"
   473→                  onClick={handlePublicCard}
   474→                  className="flex flex-col items-center gap-0.5 border-slate-200 py-2 text-[10px] text-slate-600 hover:border-emerald-300 hover:bg-emerald-50 hover:text-emerald-700"
   475→                >
   476→                  <ExternalLink className="h-4 w-4" />
   477→                  <span className="hidden lg:inline">Mi Tarjeta</span>
   478→                  <span className="lg:hidden">Ver</span>
   479→                </Button>
   480→              </TooltipTrigger>
   481→              <TooltipContent>Ver mi tarjeta pública</TooltipContent>
   482→            </Tooltip>
   483→          </TooltipProvider>
   484→          <TooltipProvider>
   485→            <Tooltip>
   486→              <TooltipTrigger asChild>
   487→                <Button
   488→                  variant="outline"
   489→                  size="sm"
   490→                  onClick={handleShare}
   491→                  className="flex flex-col items-center gap-0.5 border-slate-200 py-2 text-[10px] text-slate-600 hover:border-emerald-300 hover:bg-emerald-50 hover:text-emerald-700"

   627→function getRelativeTime(iso: string): string {
   628→  const now = Date.now();
   629→  const diff = now - new Date(iso).getTime();
   630→  const minutes = Math.floor(diff / (1000 * 60));
   631→  if (minutes < 1) return 'hace un momento';
   632→  if (minutes < 60) return `hace ${minutes} min`;
   633→  const hours = Math.floor(minutes / 60);
   634→  if (hours < 24) return `hace ${hours} h`;
   635→  const days = Math.floor(hours / 24);
   636→  if (days === 1) return 'ayer';
   637→  return `hace ${days} días`;
   638→}
