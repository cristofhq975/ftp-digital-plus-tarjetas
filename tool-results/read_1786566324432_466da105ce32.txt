   300→// ============================ TABLERO ============================
   301→function TableroSection({ onCreateOpen }: { onCreateOpen: () => void }) {
   302→  const currentUser = useAppStore(s => s.currentUser)!;
   303→  const cards = useCurrentUserCards();
   304→  const messages = useAppStore(s => s.messages);
   305→  const selectCard = useAppStore(s => s.selectCard);
   306→  const navigate = useAppStore(s => s.navigate);
   307→  const deleteCard = useAppStore(s => s.deleteCard);
   308→  const toggleCardActive = useAppStore(s => s.toggleCardActive);
   309→
   310→  const plan = PLANS[currentUser.plan];
   311→  const totalViews = cards.reduce((sum, c) => sum + c.views, 0);
   312→  const totalQrScans = cards.reduce((sum, c) => sum + c.qrScans, 0);
   313→  const unreadCount = messages.filter(m => !m.read).length;
   314→  const canCreateMore = cards.length < plan.maxCards;
   315→
   316→  const handleEdit = (card: BusinessCard) => {
   317→    selectCard(card.id);
   318→    navigate('editor');
   319→  };
   320→  const handleView = (card: BusinessCard) => {
   321→    selectCard(card.id);
   322→    navigate('public-card');
   323→  };
   324→  const handleCopy = (card: BusinessCard) => {
   325→    const link = `ftpdigitalplus.com/t/${card.linkName}`;
   326→    navigator.clipboard?.writeText(link);
   327→    toast.success('Enlace copiado', { description: link });
   328→  };
   329→  const handleDelete = (card: BusinessCard) => {
   330→    deleteCard(card.id);
   331→    toast.success(`Tarjeta "${card.cardName}" eliminada`);
   332→  };
   333→  const handleUpgrade = () => navigate('pricing');
   334→
   335→  const stats = [
   336→    {
   337→      label: 'Total de tarjetas',
   338→      value: cards.length,
   339→      sub: `de ${plan.maxCards} disponibles`,
   340→      icon: CreditCard,
   341→      color: 'emerald',
   342→      gradient: 'from-emerald-500 to-emerald-700',
   343→    },
   344→    {
   345→      label: 'Total de visitas',
   346→      value: totalViews,
   347→      sub: 'en todas las tarjetas',
   348→      icon: Eye,
   349→      color: 'amber',
   350→      gradient: 'from-amber-500 to-orange-600',
   351→    },
   352→    {
   353→      label: 'Escaneos QR',
   354→      value: totalQrScans,
   355→      sub: 'total acumulado',
   356→      icon: QrCode,
   357→      color: 'emerald',
   358→      gradient: 'from-teal-500 to-emerald-600',
   359→    },
   360→    {
   361→      label: 'Mensajes sin leer',
   362→      value: unreadCount,
   363→      sub: unreadCount > 0 ? 'requieren atención' : 'todo al día',
   364→      icon: Mail,
   365→      color: unreadCount > 0 ? 'rose' : 'emerald',
   366→      gradient: unreadCount > 0 ? 'from-rose-500 to-pink-600' : 'from-slate-500 to-slate-700',
   367→    },
   368→  ];
   369→
   370→  return (
   371→    <div className="space-y-6">
   372→      {/* Welcome header */}
   373→      <div className="relative overflow-hidden rounded-2xl bg-gradient-to-br from-emerald-600 via-emerald-700 to-emerald-800 p-6 text-white shadow-xl md:p-8">
   374→        <div className="absolute right-0 top-0 h-48 w-48 rounded-full bg-amber-400/20 blur-3xl" />
   375→        <div className="absolute -bottom-12 -left-12 h-40 w-40 rounded-full bg-emerald-400/30 blur-2xl" />
   376→        <div className="relative">
   377→          <div className="flex flex-wrap items-center justify-between gap-4">
   378→            <div>
   379→              <p className="text-sm font-medium text-emerald-100">¡Bienvenido de vuelta!</p>
   380→              <h1 className="mt-1 text-2xl font-bold md:text-3xl">
   381→                Hola, {currentUser.name.split(' ')[0]} <span className="inline-block animate-[wave_1.5s_ease-in-out_infinite] origin-[70%_70%]">👋</span>
   382→              </h1>
   383→              <p className="mt-2 max-w-lg text-sm text-emerald-100">
   384→                Gestiona tus tarjetas digitales, revisa estadísticas y mantén tu presencia al día.
   385→              </p>
   386→            </div>
   387→            <div className="flex items-center gap-2 rounded-full bg-white/15 px-3 py-1.5 text-sm font-semibold backdrop-blur">
   388→              {currentUser.plan === 'pro' && <Crown className="h-4 w-4 text-amber-300" />}
   389→              {currentUser.plan === 'basico' && <Zap className="h-4 w-4 text-amber-300" />}
   390→              {currentUser.plan === 'gratis' && <Sparkles className="h-4 w-4 text-amber-300" />}
   391→              Plan {plan.name}
   392→            </div>
   393→          </div>

   752→  const reset = () => {

   754→    setCardName('');

   772→      setError('Ese enlace ya está en uso. Prueba con otro.');
   773→      return;
   774→    }

   776→    const newId = createCard(trimmedLink, trimmedName);

   778→      setError('No se pudo crear la tarjeta. Límite alcanzado.');
   779→      return;
   780→    }

   783→    onOpenChange(false);
   784→    selectCard(newId);
