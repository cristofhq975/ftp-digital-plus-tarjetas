  1604→function LiveDemoButton() {

  1606→  const [visible, setVisible] = useState(false);
  1607→

  1610→      setVisible(window.scrollY > 400);
  1611→    };

  1613→    onScroll();

  1618→    <AnimatePresence>

  1620→        <motion.button
