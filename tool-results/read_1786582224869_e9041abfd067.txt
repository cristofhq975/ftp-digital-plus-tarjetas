   296→function StarRating({ rating, reviews, size = 'sm' }: { rating: number; reviews?: number; size?: 'sm' | 'md' }) {
   297→  const starSize = size === 'sm' ? 'h-3.5 w-3.5' : 'h-4 w-4';
   298→  return (
   299→    <div className="flex items-center gap-1.5">
   300→      <div className="flex items-center gap-0.5">
   301→        {Array.from({ length: 5 }).map((_, i) => (
   302→          <Star
   303→            key={i}
   304→            className={cn(
   305→              starSize,
   306→              i < Math.floor(rating)
   307→                ? 'fill-amber-400 text-amber-400'
   308→                : i < rating
   309→                ? 'fill-amber-400/50 text-amber-400'
   310→                : 'fill-muted text-muted-foreground/40'
   311→            )}
   312→          />
   313→        ))}
   314→      </div>
   315→      <span className="text-xs font-semibold text-foreground">{rating.toFixed(1)}</span>
   316→      {reviews !== undefined && (
   317→        <span className="text-xs text-muted-foreground">({reviews.toLocaleString('es-MX')})</span>
   318→      )}
   319→    </div>
   320→  );
   321→}
