   156→              onSupport={handleSupport}

   159→          </SheetContent>

   161→        <div className="flex items-center gap-2">

   179→

   210→                transition={{ duration: 0.25 }}

   213→                  <TableroSection
   214→                    onCreateOpen={() => setCreateOpen(true)}
   215→                    onShareCard={openShare}
   216→                  />
   217→                )}
   218→                {activeSection === 'messages' && <MessagesSection />}
   219→                {activeSection === 'appointments' && <AppointmentsSection />}

   257→// ============================ SIDEBAR ============================
   258→function SidebarContent({
   259→  user, cardsCount, maxCards, unreadCount, activeSection, onNavigate, onLogout, onSupport,
   260→  onShareCard,
   261→}: {
   262→  user: { name: string; email: string; plan: PlanType; id?: string };
   263→  cardsCount: number;
   264→  maxCards: number;
   265→  unreadCount: number;
   266→  activeSection: SectionId;
   267→  onNavigate: (id: SectionId) => void;
   268→  onLogout: () => void;
   269→  onSupport: () => void;
   270→  onShareCard: (card: BusinessCard | null) => void;
   271→}) {
   272→  const planColors = PLAN_COLORS[user.plan];
   273→  const plan = PLANS[user.plan];
   274→  const cardsUsagePercent = Math.min(100, (cardsCount / maxCards) * 100);
   275→  // Storage is approximated for the ring (we use plan.storage and assume 35% used as a demo)
   276→  const storageUsedPct = Math.min(100, 35);
   277→  const primaryCard = useAppStore(s => {
   278→    const all = s.cards;
   279→    return all.find(c => c.userId === user.id) || null;
   280→  });
   281→  const favoriteIds = useAppStore(s => s.favoriteCardIds);
   282→
   283→  // SVG circular progress ring geometry
   284→  const ringSize = 64;
   285→  const ringStroke = 6;
   286→  const radius = (ringSize - ringStroke) / 2;
   287→  const circumference = 2 * Math.PI * radius;
   288→  const cardsDash = (cardsUsagePercent / 100) * circumference;
   289→
   290→  const handlePublicCard = () => {
   291→    if (primaryCard) {
   292→      useAppStore.getState().selectCard(primaryCard.id);
   293→      useAppStore.getState().navigate('public-card');
   294→    } else {
   295→      toast.info('No tienes tarjetas para mostrar');
   296→    }
   297→  };
   298→  const handleShare = () => {
   299→    onShareCard(primaryCard);
   300→  };
   301→  const handleHelp = () => {
   302→    toast.info('Centro de ayuda', { description: 'Contáctanos en soporte@ftpdigitalplus.com' });
   303→  };
   304→
   305→  return (
   306→    <div className="flex h-full flex-col">
   307→      {/* Logo + theme toggle */}
   308→      <div className="flex items-center justify-between border-b px-5 py-4">
   309→        <FTPLogo className="h-9 w-auto" />
   310→        <ThemeToggle className="text-slate-600 hover:bg-emerald-50 hover:text-emerald-700" />
   311→      </div>
   312→
   313→      {/* User info + plan progress ring */}
   314→      <div className="border-b px-5 py-4">
   315→        <div className="flex items-center gap-3">
   316→          <div className="relative">
   317→            <Avatar className="h-11 w-11 ring-2 ring-emerald-200">
   318→              <AvatarFallback className="bg-gradient-to-br from-emerald-500 to-emerald-700 text-sm font-bold text-white">
   319→                {user.name.split(' ').map(n => n[0]).slice(0, 2).join('').toUpperCase()}
   320→              </AvatarFallback>
   321→            </Avatar>
   322→            {/* Online status indicator */}
   323→            <span
   324→              className="absolute -bottom-0.5 -right-0.5 h-3.5 w-3.5 rounded-full border-2 border-white bg-emerald-500"
   325→              aria-label="En línea"
   326→            >
   327→              <span className="absolute inset-0 animate-ping rounded-full bg-emerald-400 opacity-75" />
   328→            </span>
   329→          </div>
   330→          <div className="min-w-0 flex-1">
   331→            <p className="truncate text-sm font-semibold text-slate-800">{user.name}</p>
   332→            <p className="truncate text-xs text-muted-foreground">{user.email}</p>
   333→          </div>
   334→        </div>
   335→
   336→        {/* Plan progress ring + usage */}
   337→        <div className="mt-4 flex items-center gap-3 rounded-xl bg-gradient-to-br from-slate-50 to-emerald-50/40 p-3 ring-1 ring-slate-200/60">
   338→          {/* Circular progress ring (SVG) */}
   339→          <div className="relative" style={{ width: ringSize, height: ringSize }}>
   340→            <svg width={ringSize} height={ringSize} className="-rotate-90">
   341→              <circle
   342→                cx={ringSize / 2}
   343→                cy={ringSize / 2}
   344→                r={radius}
   345→                fill="none"
   346→                stroke="currentColor"
   347→                strokeWidth={ringStroke}
   348→                className="text-slate-200"
   349→              />
   350→              <circle
   351→                cx={ringSize / 2}
   352→                cy={ringSize / 2}
   353→                r={radius}
   354→                fill="none"
   355→                stroke="url(#planGrad)"
   356→                strokeWidth={ringStroke}
   357→                strokeDasharray={`${cardsDash} ${circumference}`}
   358→                strokeLinecap="round"
   359→                className="transition-all duration-500"
   360→              />
   361→              <defs>
   362→                <linearGradient id="planGrad" x1="0" y1="0" x2="1" y2="1">
   363→                  <stop offset="0%" stopColor="#059669" />
   364→                  <stop offset="100%" stopColor="#f59e0b" />
   365→                </linearGradient>
   366→              </defs>
   367→            </svg>
   368→            <div className="absolute inset-0 flex items-center justify-center">
   369→              <span className="text-xs font-bold text-slate-700">{cardsUsagePercent.toFixed(0)}%</span>
   370→            </div>
   371→          </div>
   372→          <div className="min-w-0 flex-1">
   373→            <p className="text-xs font-semibold text-slate-700">Uso del plan</p>
   374→            <p className="text-[11px] text-muted-foreground">
   375→              <CreditCard className="mr-1 inline h-3 w-3" />
   376→              {cardsCount} / {maxCards} tarjetas
   377→            </p>
   378→            <p className="text-[11px] text-muted-foreground">
   379→              <Database className="mr-1 inline h-3 w-3" />
   380→              {storageUsedPct}% de {plan.storage} MB

   382→          </div>
   383→        </div>

   385→        {/* Plan badge */}

   389→          {user.plan === 'gratis' && <Sparkles className="h-3 w-3" />}

   391→        </div>
   392→      </div>
   393→
   394→      {/* Nav */}
   395→      <nav className="flex-1 overflow-y-auto px-3 py-4 custom-scrollbar">
   396→        <p className="px-3 pb-2 text-[10px] font-bold uppercase tracking-wider text-muted-foreground/70">
   397→          Menú Principal
   398→        </p>
   399→        <ul className="space-y-1">
   400→          {DASHBOARD_SECTIONS.map((section) => {
   401→            const isActive = activeSection === section.id;
   402→            const isMessages = section.id === 'messages';
   403→            return (
   404→              <li key={section.id}>
   405→                <button
   406→                  onClick={() => onNavigate(section.id as SectionId)}

   408→                    'group flex w-full items-center gap-3 rounded-lg px-3 py-2.5 text-sm font-medium transition-all',
   409→                    isActive
   410→                      ? 'bg-emerald-600 text-white shadow-sm shadow-emerald-600/20'
   411→                      : 'text-slate-600 hover:bg-emerald-50 hover:text-emerald-700'
   412→                  )}
   413→                >
   414→                  <DynamicIcon name={section.icon} className={cn('h-[18px] w-[18px] shrink-0', isActive ? 'text-white' : 'text-slate-500 group-hover:text-emerald-600')} />
   415→                  <span className="flex-1 text-left">{section.name}</span>
   416→                  {isMessages && unreadCount > 0 && (
   417→                    <span className={cn(
   418→                      'rounded-full px-1.5 py-0.5 text-[10px] font-bold',
   419→                      isActive ? 'bg-white text-emerald-700' : 'bg-amber-500 text-white'
   420→                    )}>
   421→                      {unreadCount}
   422→                    </span>

   424→                  {section.id === 'tablero' && favoriteIds.length > 0 && !isActive && (
   425→                    <Star className="h-3.5 w-3.5 fill-amber-400 text-amber-400" />

   427→                  {isActive && <ChevronRight className="h-4 w-4" />}
   428→                </button>
   429→              </li>
   430→            );
   431→          })}
   432→        </ul>
   433→      </nav>
   434→
   435→      {/* Quick action buttons */}
   436→      <div className="border-t px-3 py-3">
   437→        <p className="px-2 pb-2 text-[10px] font-bold uppercase tracking-wider text-muted-foreground/70">
   438→          Acciones Rápidas
   439→        </p>
   440→        <div className="grid grid-cols-3 gap-1.5">
   441→          <TooltipProvider>
   442→            <Tooltip>
   443→              <TooltipTrigger asChild>
   444→                <Button
   445→                  variant="outline"
   446→                  size="sm"
   447→                  onClick={handlePublicCard}
   448→                  className="flex flex-col items-center gap-0.5 border-slate-200 py-2 text-[10px] text-slate-600 hover:border-emerald-300 hover:bg-emerald-50 hover:text-emerald-700"
   449→                >
   450→                  <ExternalLink className="h-4 w-4" />
   451→                  <span className="hidden lg:inline">Mi Tarjeta</span>
   452→                  <span className="lg:hidden">Ver</span>
   453→                </Button>
   454→              </TooltipTrigger>
   455→              <TooltipContent>Ver mi tarjeta pública</TooltipContent>
   456→            </Tooltip>
   457→          </TooltipProvider>
   458→          <TooltipProvider>
   459→            <Tooltip>
   460→              <TooltipTrigger asChild>
   461→                <Button
   462→                  variant="outline"
   463→                  size="sm"
   464→                  onClick={handleShare}
   465→                  className="flex flex-col items-center gap-0.5 border-slate-200 py-2 text-[10px] text-slate-600 hover:border-emerald-300 hover:bg-emerald-50 hover:text-emerald-700"
   466→                >
   467→                  <Share2 className="h-4 w-4" />

   479→                  size="sm"

   481→                  className="flex flex-col items-center gap-0.5 border-slate-200 py-2 text-[10px] text-slate-600 hover:border-emerald-300 hover:bg-emerald-50 hover:text-emerald-700"
   482→                >

   597→function getRelativeTime(iso: string): string {
   598→  const now = Date.now();
   599→  const diff = now - new Date(iso).getTime();
   600→  const minutes = Math.floor(diff / (1000 * 60));
   601→  if (minutes < 1) return 'hace un momento';
   602→  if (minutes < 60) return `hace ${minutes} min`;
   603→  const hours = Math.floor(minutes / 60);
   604→  if (hours < 24) return `hace ${hours} h`;
   605→  const days = Math.floor(hours / 24);
   606→  if (days === 1) return 'ayer';
   607→  return `hace ${days} días`;
   608→}
