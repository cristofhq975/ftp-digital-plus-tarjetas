   765→export function HelpCenter() {
   766→  const navigate = useAppStore(s => s.navigate);
   767→  const [search, setSearch] = useState('');
   768→  const [selectedGuide, setSelectedGuide] = useState<GuideItem | null>(null);
   769→  const [dialogOpen, setDialogOpen] = useState(false);
   770→  const guidesRef = useRef<HTMLDivElement>(null);
   771→
   772→  const filteredFAQ = useMemo(() => {
   773→    if (!search.trim()) return FAQ_ITEMS;
   774→    const q = search.toLowerCase();
   775→    return FAQ_ITEMS.filter((it) =>
   776→      it.question.toLowerCase().includes(q) ||
   777→      it.answer.toLowerCase().includes(q) ||
   778→      it.category.toLowerCase().includes(q)
   779→    );
   780→  }, [search]);
   781→
   782→  const handleOpenGuide = (g: GuideItem) => {
   783→    setSelectedGuide(g);
   784→    setDialogOpen(true);
   785→  };
   786→
   787→  const handleNavigateSupport = () => {
   788→    navigate('support');
   789→    toast.info('Te llevamos a soporte', {
   790→      description: 'Crea un ticket y te responderemos en menos de 24 horas.',
   791→    });
   792→  };
   793→
   794→  return (
   795→    <div className="flex min-h-screen flex-col bg-gradient-to-b from-white via-emerald-50/30 to-white dark:from-slate-950 dark:via-emerald-950/10 dark:to-slate-950">
   796→      <HeaderBar />
   797→
   798→      <main className="flex-1">
   799→        <HeroSection
   800→          search={search}
   801→          setSearch={setSearch}
   802→          resultCount={filteredFAQ.length}
   803→        />
   804→
   805→        <QuickActions onNavigateSupport={handleNavigateSupport} />
   806→
   807→        <div ref={guidesRef}>
   808→          <GuidesSection onOpenGuide={handleOpenGuide} />
   809→        </div>
   810→
   811→        <ResourcesSection />
   812→
   813→        <FAQSection items={filteredFAQ} />
   814→
   815→        <ContactCTA onNavigateSupport={handleNavigateSupport} />
   816→      </main>
   817→
   818→      <Footer />
   819→
   820→      <GuideDialog
   821→        guide={selectedGuide}
   822→        open={dialogOpen}
   823→        onOpenChange={setDialogOpen}
   824→      />
   825→    </div>
   826→  );
   827→}
