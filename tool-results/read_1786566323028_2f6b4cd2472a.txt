   124→/** Campo de carga de imagen con preview. */
   125→function ImageUpload({
   126→  label, value, onChange, aspect = 'square', description,
   127→}: {
   128→  label: string;
   129→  value: string;
   130→  onChange: (base64: string) => void;
   131→  aspect?: 'square' | 'wide' | 'circle';
   132→  description?: string;
   133→}) {
   134→  const inputRef = useRef<HTMLInputElement>(null);
   135→
   136→  const handleFile = async (e: React.ChangeEvent<HTMLInputElement>) => {
   137→    const file = e.target.files?.[0];
   138→    if (!file) return;
   139→    if (!file.type.startsWith('image/')) {
   140→      toast.error('Solo se permiten archivos de imagen');
   141→      return;
   142→    }
   143→    if (file.size > 5 * 1024 * 1024) {
   144→      toast.error('La imagen no debe superar los 5 MB');
   145→      return;
   146→    }
   147→    try {
   148→      const base64 = await fileToBase64(file);
   149→      onChange(base64);
   150→      toast.success('Imagen cargada correctamente');
   151→    } catch {
   152→      toast.error('No se pudo cargar la imagen');
   153→    }
   154→    if (inputRef.current) inputRef.current.value = '';
   155→  };
   156→
   157→  const aspectClass =
   158→    aspect === 'wide' ? 'h-20 w-32' :
   159→    aspect === 'circle' ? 'h-20 w-20 rounded-full' :
   160→    'h-20 w-20';
   161→
   162→  return (
   163→    <div>
   164→      <Label className="text-sm font-medium">{label}</Label>
   165→      {description && <p className="mb-2 text-xs text-muted-foreground">{description}</p>}
   166→      <div className="mt-2 flex items-center gap-3">
   167→        {value ? (
   168→          <div className="relative">
   169→            <div className={cn('overflow-hidden rounded-lg border', aspectClass)}>
   170→              <img src={value} alt={label} className="h-full w-full object-cover" />
   171→            </div>
   172→            <button
   173→              type="button"
   174→              onClick={() => onChange('')}
   175→              className="absolute -right-2 -top-2 rounded-full bg-destructive p-1 text-white shadow-md hover:bg-destructive/90"
   176→              aria-label="Eliminar imagen"
   177→            >
   178→              <X className="h-3 w-3" />
   179→            </button>
   180→          </div>
   181→        ) : (
   182→          <div className={cn('flex items-center justify-center border-2 border-dashed bg-muted/30 text-muted-foreground', aspectClass)}>
   183→            <ImageIcon className="h-6 w-6" />
   184→          </div>
   185→        )}
   186→        <input ref={inputRef} type="file" accept="image/*" onChange={handleFile} className="hidden" />
   187→        <Button variant="outline" size="sm" type="button" onClick={() => inputRef.current?.click()}>
   188→          <Upload className="h-4 w-4" /> Subir imagen
   189→        </Button>
   190→      </div>
   191→    </div>
   192→  );
   193→}

  1006→
  1007→function GaleriaSection({ card, updateCard }: SectionProps) {
  1008→  const inputRef = useRef<HTMLInputElement>(null);
  1009→
  1010→  const handleAddFiles = async (e: React.ChangeEvent<HTMLInputElement>) => {
  1011→    const files = Array.from(e.target.files || []);
  1012→    if (files.length === 0) return;
  1013→    const items: GalleryItem[] = [];
  1014→    for (const file of files) {
  1015→      const isVideo = file.type.startsWith('video/');
  1016→      const isImage = file.type.startsWith('image/');
  1017→      if (!isVideo && !isImage) continue;
  1018→      if (file.size > 10 * 1024 * 1024) {
  1019→        toast.error(`El archivo ${file.name} supera los 10 MB`);
  1020→        continue;
  1021→      }
  1022→      try {
  1023→        const url = await fileToBase64(file);
  1024→        items.push({
  1025→          id: generateId(),
  1026→          type: isImage ? 'image' : 'video',
  1027→          url,
  1028→          caption: '',
  1029→        });
  1030→      } catch {
  1031→        toast.error(`No se pudo cargar ${file.name}`);
  1032→      }
  1033→    }
  1034→    if (items.length > 0) {
  1035→      updateCard(card.id, { gallery: [...card.gallery, ...items] });
