   134→      navigate('template-gallery');
   135→      setMobileOpen(false);

   162→    setMobileOpen(false);

   164→
   165→  const handleSupport = () => {

   195→              onSupport={handleSupport}

   212→            <Search className="h-5 w-5" />

   215→          <NotificationsPanel />
   216→        </div>
   217→      </header>
   218→
   219→      <div className="flex flex-1">
   220→        {/* Desktop Sidebar */}
   221→        <aside className="sticky top-0 hidden h-screen w-64 shrink-0 flex-col border-r bg-white lg:w-72">
   222→          <SidebarContent
   223→            user={currentUser}
   224→            cardsCount={cards.length}
   225→            maxCards={plan.maxCards}
   226→            unreadCount={unreadCount}
   227→            activeSection={activeSection}
   228→            onNavigate={handleNavigate}
   229→            onLogout={handleLogout}
   230→            onSupport={handleSupport}
   231→            onShareCard={openShare}
   232→          />
   233→        </aside>
   234→
   235→        {/* Main */}
   236→        <main className="flex-1 overflow-x-hidden px-4 py-6 md:px-8 md:py-8">
   237→          <div className="mx-auto max-w-7xl">
   238→            {/* Pull-to-refresh hint (mobile only) */}
   239→            <div className="mb-3 flex items-center justify-center gap-1.5 text-[11px] text-muted-foreground/70 md:hidden">

   277→            Hecho con <Sparkles className="h-3 w-3 text-amber-500" /> en México
   278→          </p>

   280→      </footer>

   282→      {/* Create Card Dialog */}

   298→          toast.success('¡Tarjeta importada!', {
   299→            description: `${card.cardName} · /${card.linkName}`,
   300→          });

   314→      {/* First-time onboarding wizard */}

   320→// ============================ SIDEBAR ============================
   321→function SidebarContent({
   322→  user, cardsCount, maxCards, unreadCount, activeSection, onNavigate, onLogout, onSupport,
   323→  onShareCard,
   324→}: {
   325→  user: { name: string; email: string; plan: PlanType; id?: string };
   326→  cardsCount: number;
   327→  maxCards: number;
   328→  unreadCount: number;
   329→  activeSection: SectionId;
   330→  onNavigate: (id: SectionId) => void;
   331→  onLogout: () => void;
   332→  onSupport: () => void;
   333→  onShareCard: (card: BusinessCard | null) => void;
   334→}) {
   335→  const planColors = PLAN_COLORS[user.plan];
   336→  const plan = PLANS[user.plan];
   337→  const cardsUsagePercent = Math.min(100, (cardsCount / maxCards) * 100);
   338→  // Storage is approximated for the ring (we use plan.storage and assume 35% used as a demo)
   339→  const storageUsedPct = Math.min(100, 35);
   340→  const primaryCard = useAppStore(s => {
   341→    const all = s.cards;
   342→    return all.find(c => c.userId === user.id) || null;
   343→  });
   344→  const favoriteIds = useAppStore(s => s.favoriteCardIds);
   345→  const unreadNotifications = useAppStore(s =>
   346→    s.notifications.filter(n => !n.read).length
   347→  );
   348→
   349→  // SVG circular progress ring geometry
   350→  const ringSize = 64;
   351→  const ringStroke = 6;
   352→  const radius = (ringSize - ringStroke) / 2;
   353→  const circumference = 2 * Math.PI * radius;
   354→  const cardsDash = (cardsUsagePercent / 100) * circumference;
   355→
   356→  const handlePublicCard = () => {
   357→    if (primaryCard) {
   358→      useAppStore.getState().selectCard(primaryCard.id);
   359→      useAppStore.getState().navigate('public-card');
   360→    } else {
   361→      toast.info('No tienes tarjetas para mostrar');
   362→    }
   363→  };
   364→  const handleShare = () => {
   365→    onShareCard(primaryCard);
   366→  };
   367→  const handleHelp = () => {
   368→    toast.info('Centro de ayuda', { description: 'Contáctanos en soporte@ftpdigitalplus.com' });
   369→  };
   370→
   371→  return (
   372→    <div className="flex h-full flex-col">
   373→      {/* Logo + theme toggle */}
   374→      <div className="flex items-center justify-between border-b px-5 py-4">
   375→        <FTPLogo className="h-9 w-auto" />
   376→        <ThemeToggle className="text-slate-600 hover:bg-emerald-50 hover:text-emerald-700" />
   377→      </div>
   378→
   379→      {/* User info + plan progress ring */}
   380→      <div className="border-b px-5 py-4">
   381→        <div className="flex items-center gap-3">
   382→          <div className="relative">
   383→            <Avatar className="h-11 w-11 ring-2 ring-emerald-200">
   384→              <AvatarFallback className="bg-gradient-to-br from-emerald-500 to-emerald-700 text-sm font-bold text-white">
   385→                {user.name.split(' ').map(n => n[0]).slice(0, 2).join('').toUpperCase()}
   386→              </AvatarFallback>
   387→            </Avatar>
   388→            {/* Online status indicator */}
   389→            <span
   390→              className="absolute -bottom-0.5 -right-0.5 h-3.5 w-3.5 rounded-full border-2 border-white bg-emerald-500"
   391→              aria-label="En línea"
   392→            >
   393→              <span className="absolute inset-0 animate-ping rounded-full bg-emerald-400 opacity-75" />
   394→            </span>
   395→          </div>
   396→          <div className="min-w-0 flex-1">
   397→            <p className="truncate text-sm font-semibold text-slate-800">{user.name}</p>
   398→            <p className="truncate text-xs text-muted-foreground">{user.email}</p>
   399→          </div>
   400→        </div>
   401→
   402→        {/* Plan progress ring + usage */}
   403→        <div className="mt-4 flex items-center gap-3 rounded-xl bg-gradient-to-br from-slate-50 to-emerald-50/40 p-3 ring-1 ring-slate-200/60">
   404→          {/* Circular progress ring (SVG) */}
   405→          <div className="relative" style={{ width: ringSize, height: ringSize }}>
   406→            <svg width={ringSize} height={ringSize} className="-rotate-90">
   407→              <circle
   408→                cx={ringSize / 2}
   409→                cy={ringSize / 2}
   410→                r={radius}
   411→                fill="none"
   412→                stroke="currentColor"
   413→                strokeWidth={ringStroke}
   414→                className="text-slate-200"
   415→              />
   416→              <circle
   417→                cx={ringSize / 2}
   418→                cy={ringSize / 2}
   419→                r={radius}
   420→                fill="none"
   421→                stroke="url(#planGrad)"
   422→                strokeWidth={ringStroke}
   423→                strokeDasharray={`${cardsDash} ${circumference}`}
   424→                strokeLinecap="round"
   425→                className="transition-all duration-500"
   426→              />
   427→              <defs>
   428→                <linearGradient id="planGrad" x1="0" y1="0" x2="1" y2="1">
   429→                  <stop offset="0%" stopColor="#059669" />
   430→                  <stop offset="100%" stopColor="#f59e0b" />
   431→                </linearGradient>
   432→              </defs>
   433→            </svg>
   434→            <div className="absolute inset-0 flex items-center justify-center">
   435→              <span className="text-xs font-bold text-slate-700">{cardsUsagePercent.toFixed(0)}%</span>
   436→            </div>
   437→          </div>
   438→          <div className="min-w-0 flex-1">
   439→            <p className="text-xs font-semibold text-slate-700">Uso del plan</p>
   440→            <p className="text-[11px] text-muted-foreground">
   441→              <CreditCard className="mr-1 inline h-3 w-3" />
   442→              {cardsCount} / {maxCards} tarjetas
   443→            </p>
   444→            <p className="text-[11px] text-muted-foreground">
   445→              <Database className="mr-1 inline h-3 w-3" />
   446→              {storageUsedPct}% de {plan.storage} MB
   447→            </p>
   448→          </div>
   449→        </div>
   450→
   451→        {/* Plan badge */}
   452→        <div className={cn('mt-3 inline-flex items-center gap-1 rounded-full px-2.5 py-1 text-xs font-semibold ring-1', planColors.bg, planColors.text, planColors.ring)}>
   453→          {user.plan === 'pro' && <Crown className="h-3 w-3" />}
   454→          {user.plan === 'basico' && <Zap className="h-3 w-3" />}
   455→          {user.plan === 'gratis' && <Sparkles className="h-3 w-3" />}
   456→          Plan {planColors.label}
   457→        </div>
   458→      </div>
   459→
   460→      {/* Nav */}
   461→      <nav className="flex-1 overflow-y-auto px-3 py-4 custom-scrollbar">
   462→        <p className="px-3 pb-2 text-[10px] font-bold uppercase tracking-wider text-muted-foreground/70">
   463→          Menú Principal
   464→        </p>
   465→        <ul className="space-y-1">
   466→          {DASHBOARD_SECTIONS.map((section) => {
   467→            const isActive = activeSection === section.id;
   468→            const isMessages = section.id === 'messages';
   469→            const isNotifications = section.id === 'notifications';
   470→            return (
   471→              <li key={section.id}>
   472→                <button
   473→                  onClick={() => onNavigate(section.id as SectionId)}
   474→                  className={cn(
   475→                    'group flex w-full items-center gap-3 rounded-lg px-3 py-2.5 text-sm font-medium transition-all',
   476→                    isActive
   477→                      ? 'bg-emerald-600 text-white shadow-sm shadow-emerald-600/20'
   478→                      : 'text-slate-600 hover:bg-emerald-50 hover:text-emerald-700'
   479→                  )}
   480→                >
   481→                  <DynamicIcon name={section.icon} className={cn('h-[18px] w-[18px] shrink-0', isActive ? 'text-white' : 'text-slate-500 group-hover:text-emerald-600')} />
   482→                  <span className="flex-1 text-left">{section.name}</span>
   483→                  {isMessages && unreadCount > 0 && (
   484→                    <span className={cn(
   485→                      'rounded-full px-1.5 py-0.5 text-[10px] font-bold',
   486→                      isActive ? 'bg-white text-emerald-700' : 'bg-amber-500 text-white'
   487→                    )}>
   488→                      {unreadCount}
   489→                    </span>
   490→                  )}
   491→                  {isNotifications && unreadNotifications > 0 && (
   492→                    <span className={cn(
   493→                      'rounded-full px-1.5 py-0.5 text-[10px] font-bold',
   494→                      isActive ? 'bg-white text-emerald-700' : 'bg-rose-500 text-white'
   495→                    )}>
   496→                      {unreadNotifications > 9 ? '9+' : unreadNotifications}

   676→function getRelativeTime(iso: string): string {
   677→  const now = Date.now();
   678→  const diff = now - new Date(iso).getTime();
   679→  const minutes = Math.floor(diff / (1000 * 60));
   680→  if (minutes < 1) return 'hace un momento';
   681→  if (minutes < 60) return `hace ${minutes} min`;
   682→  const hours = Math.floor(minutes / 60);
   683→  if (hours < 24) return `hace ${hours} h`;
   684→  const days = Math.floor(hours / 24);
   685→  if (days === 1) return 'ayer';
   686→  return `hace ${days} días`;
   687→}
