    58→/** Genera datos de series temporales (mock, deterministas por día). */
    59→function generateTimeSeriesData(days: number): { date: string; views: number; scans: number }[] {
    60→  const data: { date: string; views: number; scans: number }[] = [];
    61→  const today = new Date();
    62→  for (let i = days - 1; i >= 0; i--) {
    63→    const d = new Date(today);
    64→    d.setDate(today.getDate() - i);
    65→    const seed = Math.floor(d.getTime() / 86400000);
    66→    const wave = Math.sin(seed * 0.7);
    67→    const wave2 = Math.cos(seed * 0.5);
    68→    // Escala los datos según el rango para que se vean realistas
    69→    const factor = days <= 7 ? 1 : days <= 30 ? 1.4 : 2.2;
    70→    const baseViews = Math.max(3, Math.floor((15 + wave * 9 + (seed % 14)) * factor));
    71→    const baseScans = Math.max(1, Math.floor((5 + wave2 * 4 + (seed % 8)) * factor));
    72→    data.push({
    73→      date: d.toLocaleDateString('es-MX', { day: '2-digit', month: '2-digit' }),
    74→      views: baseViews,
    75→      scans: baseScans,
    76→    });
    77→  }
    78→  return data;
    79→}
