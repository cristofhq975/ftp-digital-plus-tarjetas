   552→    (set, get) => ({

   556→      selectedCardId: null,

   566→      compareCardIds: [],
   567→      notifications: DEMO_NOTIFICATIONS,
   568→
   569→      login: (email, password) => {
   570→        const user = get().users.find(
   571→          u => u.email.toLowerCase() === email.toLowerCase() && u.password === password
   572→        );

   574→          const { password: _, ...userWithoutPassword } = user;

   578→        return false;
   579→      },

   586→

   588→

   590→
   591→      setSelectedPlanForCheckout: (plan) => set({ selectedPlanForCheckout: plan }),
   592→
   593→      createCard: (linkName, cardName) => {
   594→        const user = get().currentUser;
   595→        if (!user) return null;
   596→        const userCards = get().cards.filter(c => c.userId === user.id);
   597→        if (userCards.length >= PLANS[user.plan].maxCards) return null;
   598→        const newCard = createDefaultCard(user.id, user.plan, linkName, cardName);
   599→        set(state => ({ cards: [...state.cards, newCard], selectedCardId: newCard.id }));
   600→        return newCard.id;

   602→
   603→      updateCard: (cardId, updates) => {
   604→        set(state => ({
   605→          cards: state.cards.map(c => c.id === cardId ? { ...c, ...updates } : c),
   606→        }));
   607→      },
   608→
   609→      deleteCard: (cardId) => {
   610→        set(state => ({ cards: state.cards.filter(c => c.id !== cardId) }));
   611→      },
   612→
   613→      toggleCardActive: (cardId) => {
   614→        set(state => ({
   615→          cards: state.cards.map(c =>
   616→            c.id === cardId ? { ...c, isActive: !c.isActive } : c
   617→          ),
   618→        }));
   619→      },
   620→
   621→      generateQr: (cardId) => {
   622→        const card = get().cards.find(c => c.id === cardId);
   623→        if (!card) return;
   624→        const user = get().users.find(u => u.id === card.userId);
   625→        const plan = user?.plan || 'gratis';
   626→        if (PLANS[plan].qrExpires) {
   627→          const { generatedAt, expiresAt } = generateQrExpiration();
   628→          get().updateCard(cardId, { qrGeneratedAt: generatedAt, qrExpiresAt: expiresAt });
   629→        } else {
   630→          get().updateCard(cardId, { qrGeneratedAt: new Date().toISOString(), qrExpiresAt: null });

   659→      },
   660→
   661→      markMessageRead: (messageId) => {
   662→        set(state => ({
   663→          messages: state.messages.map(m =>
   664→            m.id === messageId ? { ...m, read: true } : m
   665→          ),
   666→        }));
   667→      },
   668→
   669→      addAppointment: (appt) => {
   670→        set(state => ({
   671→          appointments: [...state.appointments, { ...appt, id: generateId() }],
   672→        }));
   673→      },
   674→
   675→      upgradePlan: (plan) => {
   676→        const user = get().currentUser;
   677→        if (!user) return;
   678→        set(state => ({
   679→          users: state.users.map(u =>
   680→            u.id === user.id ? { ...u, plan } : u
   681→          ),
   682→          currentUser: { ...user, plan },
   683→        }));
   684→      },
   685→
   686→      updateUser: (updates) => {
   687→        const user = get().currentUser;
   688→        if (!user) return;
   689→        const updatedUser = { ...user, ...updates };
   690→        set(state => ({
   691→          users: state.users.map(u =>
   692→            u.id === user.id ? { ...u, ...updates } : u
   693→          ),
   694→          currentUser: updatedUser,
   695→        }));
   696→      },
   697→
   698→      toggleFavorite: (cardId) => {
   699→        set(state => ({
   700→          favoriteCardIds: state.favoriteCardIds.includes(cardId)
   701→            ? state.favoriteCardIds.filter(id => id !== cardId)
   702→            : [...state.favoriteCardIds, cardId],
   703→        }));
   704→      },
   705→
   706→      addTicket: (ticket) => {
   707→        const newTicket: SupportTicket = {
   708→          ...ticket,
   709→          id: generateId(),
   710→          status: 'abierto',
   711→          createdAt: new Date().toISOString(),
   712→          responses: [],
   713→        };
   714→        set(state => ({ supportTickets: [newTicket, ...state.supportTickets] }));
   715→        return newTicket.id;
   716→      },
   717→
   718→      addTicketResponse: (ticketId, response, status) => {
   719→        set(state => ({
   720→          supportTickets: state.supportTickets.map(t =>
   721→            t.id === ticketId
   722→              ? {
   723→                  ...t,
   724→                  status: status ?? t.status,
   725→                  responses: [...t.responses, response],
   726→                }
   727→              : t
   728→          ),
   729→        }));
   730→      },
   731→
   732→      setTourActive: (active) => set({ tourActive: active }),
   733→
   734→      setSelectedBlogPost: (postId) => set({ selectedBlogPost: postId }),
   735→
   736→      setCompareCards: (cardIds) => set({ compareCardIds: cardIds }),
   737→
   738→      markNotificationRead: (id) => {
   739→        set(state => ({
   740→          notifications: state.notifications.map(n =>
   741→            n.id === id ? { ...n, read: true } : n
   742→          ),
   743→        }));
   744→      },
   745→
   746→      markAllNotificationsRead: () => {
   747→        set(state => ({
   748→          notifications: state.notifications.map(n => ({ ...n, read: true })),
   749→        }));
   750→      },
   751→
   752→      deleteNotification: (id) => {
   753→        set(state => ({
   754→          notifications: state.notifications.filter(n => n.id !== id),
   755→        }));
   756→      },
   757→
   758→      addNotification: (notification) => {
   759→        const newNotification: AppNotification = {
   760→          ...notification,
   761→          id: generateId(),
   762→        };
   763→        set(state => ({ notifications: [newNotification, ...state.notifications] }));
   764→      },
   765→
   766→      // Importa una tarjeta desde un archivo JSON externo. Genera un nuevo ID,
   767→      // la asocia al usuario actual, y verifica que el plan permita más tarjetas.
   768→      // Devuelve el nuevo ID o `null` si el límite fue alcanzado.
   769→      importCard: (card) => {
   770→        const user = get().currentUser;
   771→        if (!user) return null;
   772→        const userCards = get().cards.filter(c => c.userId === user.id);
   773→        if (userCards.length >= PLANS[user.plan].maxCards) {
   774→          return null;
   775→        }
   776→        // Generar un nuevo ID único y sanitizar el linkName para evitar duplicados.
   777→        const newId = generateId();
   778→        let newLinkName = (card.linkName || '').trim().toLowerCase().replace(/[^a-z0-9-]/g, '-');
   779→        if (newLinkName.length < 3) newLinkName = `import-${newId.substring(0, 6)}`;
   780→        // Evitar colisión de linkName con tarjetas existentes
   781→        const existingLinkNames = new Set(get().cards.map(c => c.linkName));
   782→        let attempt = 0;
   783→        let finalLinkName = newLinkName;
   784→        while (existingLinkNames.has(finalLinkName) && attempt < 50) {
   785→          attempt++;
   786→          finalLinkName = `${newLinkName}-${attempt}`;
   787→        }
   788→        const importedCard: BusinessCard = {
   789→          ...card,
   790→          id: newId,
   791→          userId: user.id,
   792→          linkName: finalLinkName,
   793→          // Resetear stats al importar (la tarjeta "nace" en la cuenta actual)
   794→          views: 0,
   795→          qrScans: 0,
   796→          affiliateClicks: 0,
   797→          createdAt: new Date().toISOString(),
   798→          // Regenerar QR si el plan lo permite (no copiar QR de otra cuenta)
   799→          qrGeneratedAt: null,
   800→          qrExpiresAt: null,
   801→        };
   802→        set(state => ({
   803→          cards: [...state.cards, importedCard],
   804→          selectedCardId: importedCard.id,
   805→        }));
   806→        return importedCard.id;
   807→      },
   808→    }),
   809→    {
   810→      name: 'ftp-digital-plus-store',
   811→      storage: createJSONStorage(() => {
   812→        if (typeof window !== 'undefined') {
   813→          return window.localStorage;
   814→        }
   815→        return {
   816→          getItem: () => null,
   817→          setItem: () => {},
   818→          removeItem: () => {},
   819→        };
   820→      }),
   821→      partialize: (state) => ({
   822→        currentUser: state.currentUser,
   823→        users: state.users,
   824→        cards: state.cards,
   825→        messages: state.messages,
   826→        appointments: state.appointments,
   827→        favoriteCardIds: state.favoriteCardIds,
   828→        supportTickets: state.supportTickets,
   829→        selectedPlanForCheckout: state.selectedPlanForCheckout,
   830→        notifications: state.notifications,
   831→      }),
   832→    }
   833→  )
