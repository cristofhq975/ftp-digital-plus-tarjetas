    66→type CardBrand = 'visa' | 'mastercard' | 'amex' | 'desconocida';
    67→
    68→function detectBrand(rawNumber: string): CardBrand {
    69→  const n = rawNumber.replace(/\s+/g, '');
    70→  if (/^4/.test(n)) return 'visa';
    71→  if (/^3[47]/.test(n)) return 'amex';
    72→  if (/^(5[1-5]|2[2-7])/.test(n)) return 'mastercard';
    73→  return 'desconocida';
    74→}
    75→
    76→function formatCardNumber(value: string): string {
    77→  const digits = value.replace(/\D/g, '').slice(0, 16);
    78→  return digits.replace(/(.{4})/g, '$1 ').trim();
    79→}
    80→
    81→function formatExpiry(value: string): string {
    82→  const digits = value.replace(/\D/g, '').slice(0, 4);
    83→  if (digits.length <= 2) return digits;
    84→  return `${digits.slice(0, 2)}/${digits.slice(2)}`;
    85→}
    86→
    87→function formatCvc(value: string): string {
    88→  return value.replace(/\D/g, '').slice(0, 4);
    89→}
    90→
    91→function currency(n: number): string {
    92→  return n.toLocaleString('es-MX', { minimumFractionDigits: 2, maximumFractionDigits: 2 });
    93→}

    99→const cardSchema = z.object({
   100→  cardNumber: z
   101→    .string()
   102→    .min(1, 'El número de tarjeta es obligatorio')
   103→    .refine(v => v.replace(/\s+/g, '').length >= 13, 'Número de tarjeta incompleto')
   104→    .refine(v => v.replace(/\s+/g, '').length <= 16, 'Máximo 16 dígitos'),
   105→  expiry: z
   106→    .string()
   107→    .min(1, 'Fecha requerida')
   108→    .regex(/^\d{2}\/\d{2}$/, 'Formato MM/YY')
   109→    .refine(v => {
   110→      const [mm, yy] = v.split('/').map(Number);
   111→      if (mm < 1 || mm > 12) return false;
   112→      const now = new Date();
   113→      const currentYear = now.getFullYear() % 100;
   114→      const currentMonth = now.getMonth() + 1;
   115→      if (yy < currentYear) return false;
   116→      if (yy === currentYear && mm < currentMonth) return false;
   117→      return true;
   118→    }, 'Tarjeta vencida'),
   119→  cvc: z
   120→    .string()
   121→    .min(3, 'CVC inválido')
   122→    .max(4, 'CVC inválido'),
   123→  nameOnCard: z
   124→    .string()
   125→    .min(3, 'El nombre es demasiado corto')
   126→    .max(60, 'El nombre es demasiado largo'),
   127→  email: z
   128→    .string()
   129→    .min(1, 'Email requerido')
   130→    .email('Email inválido'),
   131→});

   865→function PaymentLayout({ plan, onBack, onSuccess }: { plan: PlanType; onBack: () => void; onSuccess: () => void }) {
   866→  const currentUser = useAppStore(s => s.currentUser);
   867→  const emailDefault = currentUser?.email ?? '';
   868→
   869→  const [method, setMethod] = useState<PaymentMethod>('tarjeta');
   870→  const [acceptTerms, setAcceptTerms] = useState(false);
   871→  const [saveCard, setSaveCard] = useState(true);
   872→  const [processing, setProcessing] = useState(false);
   873→  const [promo, setPromo] = useState<{ code: string; percent: number } | null>(null);
   874→  const [promoApplied, setPromoApplied] = useState(false);
   875→
   876→  const cardForm = useForm<CardFormValues>({
   877→    resolver: zodResolver(cardSchema),
   878→    mode: 'onTouched',
   879→    defaultValues: {
   880→      cardNumber: '',
   881→      expiry: '',
   882→      cvc: '',
   883→      nameOnCard: currentUser?.name ?? '',
   884→      email: emailDefault,
   885→    },
   886→  });
   887→
   888→  const paypalForm = useForm<PaypalFormValues>({
   889→    resolver: zodResolver(paypalSchema),
   890→    mode: 'onTouched',
   891→    defaultValues: { email: emailDefault },
   892→  });
   893→
   894→  const transferForm = useForm<TransferFormValues>({
   895→    resolver: zodResolver(transferSchema),
   896→    mode: 'onTouched',
   897→    defaultValues: { holderName: currentUser?.name ?? '' },
   898→  });
   899→
   900→  const [cardNumber, setCardNumber] = useState('');
   901→  const [expiry, setExpiry] = useState('');
   902→  const [cvc, setCvc] = useState('');
   903→  const brand = useMemo<CardBrand>(() => detectBrand(cardNumber), [cardNumber]);
   904→
   905→  const onCardNumberChange = (e: ChangeEvent<HTMLInputElement>) => {
   906→    const formatted = formatCardNumber(e.target.value);
   907→    setCardNumber(formatted);
   908→    cardForm.setValue('cardNumber', formatted, { shouldValidate: true });
   909→  };
   910→  const onExpiryChange = (e: ChangeEvent<HTMLInputElement>) => {
   911→    const formatted = formatExpiry(e.target.value);
   912→    setExpiry(formatted);
   913→    cardForm.setValue('expiry', formatted, { shouldValidate: true });
   914→  };
   915→  const onCvcChange = (e: ChangeEvent<HTMLInputElement>) => {
   916→    const formatted = formatCvc(e.target.value);
   917→    setCvc(formatted);
   918→    cardForm.setValue('cvc', formatted, { shouldValidate: true });
   919→  };
   920→
   921→  const config = PLANS[plan];
   922→  const subtotal = config.price;
   923→  const discount = promo ? subtotal * promo.percent : 0;
   924→  const base = subtotal - discount;
   925→  const iva = base * IVA_RATE;
   926→  const total = base + iva;
   927→
   928→  const handleApplyPromo = (code: string) => {
   929→    const pct = PROMO_CODES[code];
   930→    if (!pct) {
   931→      toast.error('Código de descuento inválido', { description: `Intenta con FTP10 o BIENVENIDA` });
   932→      return;
   933→    }
   934→    setPromo({ code, percent: pct });
   935→    setPromoApplied(true);
   936→    toast.success(`Código ${code} aplicado`, { description: `${Math.round(pct * 100)}% de descuento` });
   937→  };
   938→  const handleRemovePromo = () => {
   939→    setPromo(null);
   940→    setPromoApplied(false);
   941→    toast('Código de descuento removido');
   942→  };
   943→
   944→  const isFormValid = useMemo(() => {
   945→    if (!acceptTerms) return false;
   946→    if (method === 'tarjeta') return cardForm.formState.isValid;
   947→    if (method === 'paypal') return paypalForm.formState.isValid;
   948→    return transferForm.formState.isValid;
   949→  }, [acceptTerms, method, cardForm.formState.isValid, paypalForm.formState.isValid, transferForm.formState.isValid]);
   950→
   951→  const onSubmit = async (e: FormEvent<HTMLFormElement>) => {
   952→    e.preventDefault();
   953→    if (!acceptTerms) {
   954→      toast.error('Debes aceptar los Términos y Condiciones');
   955→      return;
   956→    }
   957→
   958→    let valid = false;
   959→    if (method === 'tarjeta') {
   960→      valid = await cardForm.trigger();
   961→    } else if (method === 'paypal') {
   962→      valid = await paypalForm.trigger();
   963→    } else {
   964→      valid = await transferForm.trigger();
   965→    }
   966→    if (!valid) {
   967→      toast.error('Revisa los campos del formulario');
   968→      return;
   969→    }
   970→
   971→    setProcessing(true);
   972→    const methodLabel = method === 'tarjeta' ? 'tarjeta' : method === 'paypal' ? 'PayPal' : 'transferencia';
   973→    toast.loading(`Procesando pago vía ${methodLabel}...`, { id: 'pay-toast' });
   974→
   975→    await new Promise(resolve => setTimeout(resolve, 2000));
   976→
   977→    toast.dismiss('pay-toast');
   978→    toast.success('¡Pago completado!', {
   979→      description: `Plan ${config.name} activado correctamente.`,
   980→    });
   981→    setProcessing(false);
   982→    onSuccess();
   983→  };
   984→
   985→  return (
   986→    <div className="mx-auto max-w-7xl px-4 py-8 sm:px-6 lg:px-8">
   987→      <div className="mb-6 flex flex-col items-center gap-4">
   988→        <StepIndicator current="pago" />
   989→      </div>
   990→
   991→      <form onSubmit={onSubmit} className="grid gap-6 lg:grid-cols-5">
   992→        {/* Left: payment form (60%) */}
   993→        <div className="lg:col-span-3">
   994→          <Card className="border-slate-200 shadow-sm">
   995→            <CardHeader className="border-b border-slate-100">
   996→              <CardTitle className="text-lg text-slate-900">Información de pago</CardTitle>
   997→              <CardDescription>Elige tu método de pago preferido y completa los datos.</CardDescription>
   998→            </CardHeader>
   999→            <CardContent className="space-y-5 p-5 sm:p-6">
  1000→              <Tabs value={method} onValueChange={(v) => setMethod(v as PaymentMethod)}>
  1001→                <TabsList className="grid w-full grid-cols-3">
  1002→                  {PAYMENT_METHODS.map(m => (
  1003→                    <TabsTrigger key={m.id} value={m.id} className="flex flex-col gap-1 py-2">
  1004→                      <m.icon className="size-4" />
  1005→                      <span className="text-xs">{m.label}</span>
  1006→                    </TabsTrigger>
  1007→                  ))}
  1008→                </TabsList>
  1009→
  1010→                <TabsContent value="tarjeta" className="mt-5">
  1011→                  <CardForm
  1012→                    register={cardForm.register}
  1013→                    errors={cardForm.formState.errors}
  1014→                    cardNumber={cardNumber}
  1015→                    brand={brand}
  1016→                    onCardNumber={onCardNumberChange}
  1017→                    expiry={expiry}
  1018→                    cvc={cvc}
  1019→                    onExpiry={onExpiryChange}
  1020→                    onCvc={onCvcChange}
  1021→                    saveCard={saveCard}
  1022→                    setSaveCard={setSaveCard}
  1023→                    emailDefault={emailDefault}
  1024→                  />
  1025→                </TabsContent>
  1026→
  1027→                <TabsContent value="paypal" className="mt-5">
  1028→                  <PayPalForm
  1029→                    register={paypalForm.register}
  1030→                    errors={paypalForm.formState.errors}
  1031→                    emailDefault={emailDefault}
  1032→                  />
  1033→                </TabsContent>
  1034→
  1035→                <TabsContent value="transferencia" className="mt-5">
  1036→                  <TransferenciaForm

  1042→
  1043→              <SecurityBadges />
  1044→
  1045→              <label
  1046→                htmlFor="terms"
  1047→                className="flex cursor-pointer items-start gap-2.5 rounded-lg border border-slate-200 bg-slate-50/60 p-3 text-sm text-slate-700 transition-colors hover:bg-slate-50"
  1048→              >

  1062→                    Política de Privacidad

  1064→                  .

  1069→                type="submit"
  1070→                disabled={!isFormValid || processing}
  1071→                className="w-full bg-gradient-to-r from-emerald-600 to-emerald-500 text-white shadow-md hover:from-emerald-700 hover:to-emerald-600 disabled:cursor-not-allowed disabled:opacity-60"
