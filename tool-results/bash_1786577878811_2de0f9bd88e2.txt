     4→import { motion, AnimatePresence } from 'framer-motion';
     5→import { QRCodeCanvas } from 'qrcode.react';
     6→import { toast } from 'sonner';
     7→import {
     8→  X,
     9→  Check,
    10→  ArrowRight,
    11→  ArrowLeft,
    12→  Sparkles,
    13→  User,
    14→  QrCode,
    15→  Share2,
    16→  Copy,
    17→  Wand2,
    18→} from 'lucide-react';
    19→import { Button } from '@/components/ui/button';
    20→import { Input } from '@/components/ui/input';

    22→import { useAppStore, useCurrentUserCards } from '@/lib/store';

    27→
    28→const STORAGE_KEY = 'ftp-onboarding-completed';
    29→const TOTAL_STEPS = 4;
    30→
    31→// Read onboarding-completed flag from localStorage (client-only) via
    32→// useSyncExternalStore to avoid setState-in-effect lint warnings.
    33→const subscribeNoop = () => () => {};
    34→function readCompletedClient(): boolean {
    35→  try {
    36→    return localStorage.getItem(STORAGE_KEY) === '1';
    37→  } catch {
    38→    return false;
    39→  }
    40→}
    41→function readCompletedServer(): boolean {
    42→  return false;
    43→}
    44→
    45→export function OnboardingWizard() {
    46→  const currentUser = useAppStore(s => s.currentUser);
    47→  const createCard = useAppStore(s => s.createCard);
    48→  const cards = useCurrentUserCards();
    49→  const selectedCard = useAppStore(s => {
    50→    if (!s.selectedCardId) return null;
    51→    return s.cards.find(c => c.id === s.selectedCardId) || null;
    52→  });
    53→
    54→  const isCompleted = useSyncExternalStore(
    55→    subscribeNoop,
    56→    readCompletedClient,
    57→    readCompletedServer
    58→  );
    59→  const [dismissed, setDismissed] = useState(false);
    60→  const shouldShow = !!currentUser && !isCompleted && !dismissed;
    61→
    62→  const [step, setStep] = useState(1);
    63→  const [cardName, setCardName] = useState('');
    64→  const [linkName, setLinkName] = useState('');
    65→  const [linkTouched, setLinkTouched] = useState(false);
    66→  const [creating, setCreating] = useState(false);
    67→  const [newCardLinkName, setNewCardLinkName] = useState<string | null>(null);
    68→
    69→  const closeWizard = useCallback(() => {
    70→    try {
    71→      localStorage.setItem(STORAGE_KEY, '1');
    72→    } catch {
    73→      /* ignore */
    74→    }
    75→    setDismissed(true);
    76→  }, []);
    77→
    78→  const hasExistingCard = cards.length > 0;
    79→  const plan = currentUser ? PLANS[currentUser.plan] : null;
    80→  const atLimit = plan ? cards.length >= plan.maxCards : false;
    81→
    82→  const canCreate = useMemo(() => {
    83→    if (!cardName.trim() || !linkName.trim()) return false;
    84→    if (cards.some(c => c.linkName === linkName)) return false;
    85→    return true;
    86→  }, [cardName, linkName, cards]);
    87→
    88→  const handleCreateCard = useCallback(() => {
    89→    if (!canCreate || atLimit) return;
    90→    setCreating(true);
    91→    // Small delay for UX feedback
    92→    setTimeout(() => {
    93→      const id = createCard(linkName.trim(), cardName.trim());
    94→      setCreating(false);
    95→      if (id) {
    96→        setNewCardLinkName(linkName.trim());
    97→        toast.success('¡Tarjeta creada!', {
    98→          description: `"${cardName}" está lista para personalizar.`,
    99→        });
   100→        setStep(3);
   101→      } else {
   102→        toast.error('No se pudo crear la tarjeta', {
   103→          description: 'Es posible que hayas alcanzado el límite de tu plan.',
   104→        });
   105→      }
   106→    }, 400);
   107→  }, [canCreate, atLimit, createCard, linkName, cardName]);
   108→
   109→  const handleNext = useCallback(() => {
   110→    setStep(s => Math.min(TOTAL_STEPS, s + 1));
   111→  }, []);
   112→
   113→  const handlePrev = useCallback(() => {
   114→    setStep(s => Math.max(1, s - 1));
   115→  }, []);
   116→
   117→  const handleSkip = useCallback(() => {
   118→    toast.info('Tutorial omitido', {
   119→      description: 'Puedes completarlo más adelante si lo deseas.',
   120→    });
   121→    closeWizard();
   122→  }, [closeWizard]);
   123→
   124→  const handleFinish = useCallback(() => {
   125→    toast.success('¡Listo! Bienvenido a FTP Digital Plus', {
   126→      description: 'Ya puedes empezar a crear y compartir tus tarjetas.',
   127→    });
   128→    closeWizard();
   129→  }, [closeWizard]);
   130→
   131→  if (!shouldShow || !currentUser) return null;
   132→
   133→  // Determine which card link name to show in step 4
   134→  const shareLinkName =
   135→    newCardLinkName ||
   136→    (selectedCard?.linkName ?? null) ||
   137→    (cards[0]?.linkName ?? null);
   138→
   139→  const progress = (step / TOTAL_STEPS) * 100;
   140→
   141→  return (
   142→    <div
   143→      className="fixed inset-0 z-[100] flex items-center justify-center bg-slate-900/60 p-3 backdrop-blur-sm sm:p-6"
   144→      role="dialog"
   145→      aria-modal="true"
   146→      aria-labelledby="onboarding-title"
   147→    >
   148→      <motion.div
   149→        initial={{ opacity: 0, scale: 0.96, y: 8 }}
   150→        animate={{ opacity: 1, scale: 1, y: 0 }}
   151→        exit={{ opacity: 0, scale: 0.96, y: 8 }}
   152→        transition={{ duration: 0.25, ease: 'easeOut' }}
   153→        className="relative flex max-h-[92vh] w-full max-w-2xl flex-col overflow-hidden rounded-2xl bg-white shadow-2xl ring-1 ring-emerald-900/10"
   154→      >
   155→        {/* Top bar */}
   156→        <div className="flex items-center justify-between gap-3 border-b border-emerald-100/70 bg-gradient-to-r from-emerald-50/80 via-white to-amber-50/50 px-4 py-3 sm:px-6">
   157→          <div className="flex items-center gap-3">
   158→            <div className="flex h-8 w-8 items-center justify-center rounded-lg bg-gradient-to-br from-emerald-500 to-emerald-700 text-white shadow-sm">
   159→              <Sparkles className="h-4 w-4" />
   160→            </div>
   161→            <div>
   162→              <p className="text-xs font-medium uppercase tracking-wider text-emerald-700">
   163→                Tutorial · Paso {step} de {TOTAL_STEPS}
   164→              </p>
   165→              <p className="text-[11px] text-muted-foreground">
   166→                {step === 1 && 'Bienvenida'}
   167→                {step === 2 && 'Crea tu primera tarjeta'}
   168→                {step === 3 && 'Personaliza tu tarjeta'}
   169→                {step === 4 && 'Comparte con el mundo'}
   170→              </p>
   171→            </div>
   172→          </div>
   173→          <div className="flex items-center gap-1">
   174→            <Button
   175→              variant="ghost"
   176→              size="sm"
   177→              onClick={handleSkip}
   178→              className="text-xs text-slate-600 hover:bg-slate-100 hover:text-slate-800"
   179→            >
   180→              <span className="hidden sm:inline">Saltar tutorial</span>
   181→              <span className="sm:hidden">Saltar</span>
   182→            </Button>
   183→            <Button
   184→              variant="ghost"
   185→              size="icon"
   186→              onClick={closeWizard}
   187→              className="size-8 text-slate-500 hover:bg-slate-100 hover:text-slate-800"
   188→              aria-label="Cerrar"
   189→            >
   190→              <X className="h-4 w-4" />
   191→            </Button>
   192→          </div>
   193→        </div>
   194→
   195→        {/* Progress bar */}
   196→        <div className="h-1 w-full bg-emerald-50">
   197→          <motion.div
   198→            className="h-full bg-gradient-to-r from-emerald-500 to-amber-500"
   199→            initial={false}
   200→            animate={{ width: `${progress}%` }}
   201→            transition={{ duration: 0.4, ease: 'easeInOut' }}
   202→          />
   203→        </div>
   204→
   205→        {/* Step content */}
   206→        <div className="flex-1 overflow-y-auto">
   207→          <AnimatePresence mode="wait">
   208→            <motion.div
   209→              key={step}
   210→              initial={{ opacity: 0, x: 24 }}
   211→              animate={{ opacity: 1, x: 0 }}
   212→              exit={{ opacity: 0, x: -24 }}
   213→              transition={{ duration: 0.25, ease: 'easeOut' }}
   214→              className="px-4 py-6 sm:px-8 sm:py-8"
   215→            >
   216→              {step === 1 && <StepWelcome />}
   217→              {step === 2 && (
   218→                <StepCreateCard
   219→                  cardName={cardName}
   220→                  linkName={linkName}
   221→                  onCardNameChange={v => {
   222→                    setCardName(v);
   223→                    if (!linkTouched) setLinkName(slugify(v));
   224→                  }}
   225→                  onLinkNameChange={v => {
   226→                    setLinkTouched(true);
   227→                    setLinkName(v);
   228→                  }}
   229→                  hasExistingCard={hasExistingCard}
   230→                  existingCount={cards.length}
   231→                  atLimit={atLimit}
   232→                  linkTaken={cards.some(c => c.linkName === linkName)}
   233→                />
   234→              )}
   235→              {step === 3 && <StepPersonalize />}
   236→              {step === 4 && <StepShare linkName={shareLinkName} />}
   237→            </motion.div>
   238→          </AnimatePresence>
   239→        </div>
   240→
   241→        {/* Bottom navigation */}
   242→        <div className="flex items-center justify-between gap-3 border-t border-emerald-100/70 bg-white px-4 py-3 sm:px-8">
   243→          <div className="flex items-center gap-1.5">
   244→            {Array.from({ length: TOTAL_STEPS }).map((_, i) => (
   245→              <div
   246→                key={i}
   247→                className={cn(
   248→                  'h-1.5 rounded-full transition-all',
   249→                  i + 1 === step
   250→                    ? 'w-6 bg-emerald-600'
   251→                    : i + 1 < step
   252→                      ? 'w-1.5 bg-emerald-400'
   253→                      : 'w-1.5 bg-slate-200'
   254→                )}
   255→              />
   256→            ))}
   257→          </div>
   258→          <div className="flex items-center gap-2">
   259→            {step > 1 && step !== 4 && (
   260→              <Button
   261→                variant="outline"
   262→                size="sm"
   263→                onClick={handlePrev}
   264→                className="border-emerald-200 text-emerald-700 hover:bg-emerald-50 hover:text-emerald-800"
   265→              >
   266→                <ArrowLeft className="h-4 w-4" />
   267→                Atrás
   268→              </Button>
   269→            )}
   270→            {step === 1 && (
   271→              <Button
   272→                size="sm"
   273→                onClick={handleNext}
   274→                className="bg-gradient-to-r from-emerald-600 to-emerald-500 text-white shadow-sm hover:from-emerald-700 hover:to-emerald-600"
   275→              >
   276→                Siguiente
   277→                <ArrowRight className="h-4 w-4" />
   278→              </Button>
   279→            )}
   280→            {step === 2 && (
   281→              <>
   282→                {hasExistingCard && (
   283→                  <Button
   284→                    size="sm"
   285→                    variant="outline"
   286→                    onClick={handleNext}
   287→                    className="border-slate-200 text-slate-600 hover:bg-slate-50"
   288→                  >
   289→                    Continuar sin crear
   290→                    <ArrowRight className="h-4 w-4" />
   291→                  </Button>
   292→                )}
   293→                {!atLimit && (
   294→                  <Button
   295→                    size="sm"
   296→                    onClick={handleCreateCard}
   297→                    disabled={!canCreate || creating}
   298→                    className="bg-gradient-to-r from-emerald-600 to-emerald-500 text-white shadow-sm hover:from-emerald-700 hover:to-emerald-600 disabled:opacity-50"
   299→                  >
   300→                    {creating ? (
   301→                      <>
   302→                        <span className="h-4 w-4 animate-spin rounded-full border-2 border-white/40 border-t-white" />
   303→                        Creando…
   304→                      </>
   305→                    ) : (
   306→                      <>
   307→                        Crear y continuar
   308→                        <ArrowRight className="h-4 w-4" />
   309→                      </>
   310→                    )}
   311→                  </Button>
   312→                )}
   313→                {atLimit && !hasExistingCard && (
   314→                  <Button
   315→                    size="sm"
   316→                    onClick={handleNext}
   317→                    className="bg-gradient-to-r from-amber-500 to-amber-600 text-white shadow-sm hover:from-amber-600 hover:to-amber-700"
   318→                  >
   319→                    Continuar
   320→                    <ArrowRight className="h-4 w-4" />
   321→                  </Button>
   322→                )}
   323→              </>
   324→            )}
   325→            {step === 3 && (
   326→              <Button
   327→                size="sm"
   328→                onClick={handleNext}
   329→                className="bg-gradient-to-r from-emerald-600 to-emerald-500 text-white shadow-sm hover:from-emerald-700 hover:to-emerald-600"
   330→              >
   331→                Entendido
   332→                <ArrowRight className="h-4 w-4" />
   333→              </Button>
   334→            )}
   335→            {step === 4 && (
   336→              <Button
   337→                size="sm"
   338→                onClick={handleFinish}
   339→                className="bg-gradient-to-r from-emerald-600 to-emerald-500 text-white shadow-sm hover:from-emerald-700 hover:to-emerald-600"
   340→              >
   341→                <Check className="h-4 w-4" />
   342→                Finalizar
   343→              </Button>
   344→            )}
   345→          </div>
   346→        </div>
   347→      </motion.div>
   348→    </div>
   349→  );
   350→}
