    42→function getQrValue(card: BusinessCard, plan: PlanType): string {
    43→  const planConfig = PLANS[plan];
    44→  const expired = planConfig.qrExpires && card.qrExpiresAt ? isQrExpired(card) : false;
    45→  if (expired) return 'https://ftpdigitalplus.com/qr-expirado';
    46→  if (card.whatsappNumber) {
    47→    return buildWhatsappUrl(card.whatsappNumber, card.whatsappMessage || 'Hola, vi tu tarjeta digital');
    48→  }
    49→  return 'https://ftpdigitalplus.com';
    50→}
    51→
    52→function useCardDownload(card: BusinessCard, userPlan: PlanType) {
    53→  const qrRef = useRef<HTMLDivElement>(null);
    54→  const [downloading, setDownloading] = useState(false);
    55→
    56→  const handleDownload = async () => {
    57→    const qrCanvas = qrRef.current?.querySelector('canvas');
    58→    if (!qrCanvas) {
    59→      toast.error('No se pudo generar el código QR');
    60→      return;
    61→    }
    62→    setDownloading(true);
    63→    try {
    64→      const qrDataUrl = (qrCanvas as HTMLCanvasElement).toDataURL('image/png');
    65→      const imageDataUrl = await generateCardImage(card, userPlan, qrDataUrl);
    66→      downloadDataUrl(imageDataUrl, `tarjeta-${card.linkName}.png`);
    67→      toast.success('Tarjeta descargada correctamente');
    68→    } catch {
    69→      toast.error('Ocurrió un error al generar la imagen');
    70→    } finally {
    71→      setDownloading(false);
    72→    }
    73→  };
    74→
    75→  return { qrRef, handleDownload, downloading };
    76→}

   224→/* ---------- password gate ---------- */
   225→
   226→function PasswordGate({ card, children }: { card: BusinessCard; children: React.ReactNode }) {
   227→  const [unlocked, setUnlocked] = useState(!card.passwordProtected);
   228→  const [password, setPassword] = useState('');
   229→  const [error, setError] = useState(false);
   230→
   231→  if (unlocked) return <>{children}</>;
   232→
   233→  const handleSubmit = (e: React.FormEvent) => {
   234→    e.preventDefault();
   235→    if (password === card.cardPassword) {
   236→      setUnlocked(true);
   237→      toast.success('Acceso concedido');
   238→    } else {
   239→      setError(true);
   240→      toast.error('Contraseña incorrecta');
   241→    }
   242→  };
   243→
   244→  return (
   245→    <div className="flex min-h-[60vh] items-center justify-center">
   246→      <motion.div
   247→        initial={{ opacity: 0, y: 20 }}
   248→        animate={{ opacity: 1, y: 0 }}
   249→        transition={{ duration: 0.4 }}
   250→        className="w-full max-w-sm"
   251→      >
   252→        <Card className="border-emerald-100 shadow-xl">
   253→          <CardContent className="pt-6 text-center">
   254→            <div className="mx-auto mb-4 flex h-16 w-16 items-center justify-center rounded-full bg-emerald-100">
   255→              <Lock className="h-8 w-8 text-emerald-600" />
   256→            </div>
   257→            <h2 className="text-lg font-bold text-foreground">Tarjeta protegida</h2>
   258→            <p className="mt-1 text-sm text-muted-foreground">
   259→              Esta tarjeta está protegida con contraseña. Ingrésala para continuar.
   260→            </p>
   261→            <form onSubmit={handleSubmit} className="mt-5 space-y-3 text-left">
   262→              <div className="space-y-1.5">
   263→                <Label htmlFor="card-password" className="text-xs">Contraseña</Label>
   264→                <Input
   265→                  id="card-password"
   266→                  type="password"
   267→                  value={password}
   268→                  onChange={(e) => { setPassword(e.target.value); setError(false); }}
   269→                  placeholder="••••••••"
   270→                  autoFocus
   271→                  className={cn(error && 'border-destructive focus-visible:ring-destructive/20')}
   272→                />
   273→                {error && <p className="text-xs text-destructive">Contraseña incorrecta. Inténtalo de nuevo.</p>}
   274→              </div>
   275→              <Button type="submit" className="w-full bg-emerald-600 text-white hover:bg-emerald-700">
   276→                <Lock className="h-4 w-4" />
   277→                Desbloquear tarjeta
   278→              </Button>
   279→            </form>
   280→          </CardContent>
   281→        </Card>
   282→      </motion.div>
   283→    </div>
   284→  );
   285→}

   522→          href={`https://twitter.com/intent/tweet?text=${encodeURIComponent('Mira esta tarjeta digital')}&url=${encodeURIComponent(shareUrl)}`}

   524→          rel="noopener noreferrer"
   525→          aria-label="Compartir en Twitter / X"
   526→          className="flex h-8 w-8 items-center justify-center rounded-full border border-emerald-100 text-muted-foreground transition hover:border-emerald-300 hover:bg-emerald-50 hover:text-emerald-700"
   527→        >
   528→          <Twitter className="h-4 w-4" />
   529→        </a>
   530→      </div>
   531→
   532→      {/* Contact form + Appointment booking */}
   533→      <div className="mt-6 grid gap-4">
   534→        <ContactFormCard onAddMessage={onAddMessage} primaryColor={card.primaryColor} />
   535→        {card.team.length > 0 && (
   536→          <AppointmentBookingCard card={card} onAddAppointment={onAddAppointment} primaryColor={card.primaryColor} />
   537→        )}
   538→      </div>
   539→    </motion.div>
   540→  );
   541→}

   657→function AppointmentBookingCard({
   658→  card,
   659→  onAddAppointment,
   660→  primaryColor,
   661→}: {
   662→  card: BusinessCard;
   663→  onAddAppointment: (appt: {
   664→    teamMemberId: string;
   665→    clientName: string;
   666→    clientEmail: string;
   667→    date: string;
   668→    time: string;
   669→    status: 'pending' | 'confirmed' | 'cancelled';
   670→  }) => void;
   671→  primaryColor: string;
   672→}) {
   673→  const [open, setOpen] = useState(false);
   674→  const [teamMemberId, setTeamMemberId] = useState(card.team[0]?.id || '');
   675→  const [date, setDate] = useState('');
   676→  const [time, setTime] = useState('');
   677→  const [clientName, setClientName] = useState('');
   678→  const [clientEmail, setClientEmail] = useState('');
